## 可能的问题

### 问题 1: 模型绑定
**症状**: 设备收不到消息，串口没有任何日志

**原因**: Sensor Server 模型可能没有绑定 App Key

**检查方法**:
1. 在 nRF Mesh App 中查看设备的模型列表
2. 确认 Sensor Server (0x1100) 是否绑定了 App Key

**解决方法**:
```kotlin
// 在配置设备时，确保绑定 Sensor Server 模型
val sensorServerModelId = 0x1100
val appKey = network.appKeys.first()
configModelAppBind(deviceAddress, elementAddress, sensorServerModelId, appKey)
```

### 问题 2: 发送时机
**症状**: 偶尔能收到，但大部分时候收不到

**原因**: 在连接不稳定或网络未就绪时发送消息

**解决方法**:
```kotlin
fun readTemperature(address: Int) {
    // 检查连接状态
    if (!MeshState.bleConnection.isConnected) {
        Log.e("MeshApp", "BLE 未连接")
        return
    }
    
    // 检查网络状态
    val network = MeshState.meshNetWork ?: return
    if (!MeshState.isNetworkLoaded.value) {
        Log.e("MeshApp", "网络未加载")
        return
    }
    
    // 添加延迟，确保上一条消息已发送完成
    Handler(Looper.getMainLooper()).postDelayed({
        val appKey = network.appKeys.firstOrNull() ?: return@postDelayed
        Log.d("MeshApp", "读取温度: address=0x${address.toString(16)}")
        MeshState.meshManagerApi.createMeshPdu(address, SensorGet(appKey, null))
    }, 100)
}
```

### 问题 3: 消息超时

- 设备正在处理其他消息
- 网络拥塞

**解决方法**:

```kotlin
fun readTemperature(device: ProvisionedMeshNode) {

}
```

```kotlin
fun readTemperature(address: Int) {
    Log.d("MeshApp", "=== 读取温度 ===")
    
    // 1. 检查连接
    if (!MeshState.bleConnection.isConnected) {
        Log.e("MeshApp", "BLE 未连接")
        return
    }
    
    // 2. 检查网络
    val network = MeshState.meshNetWork ?: run {
        Log.e("MeshApp", "网络未初始化")
        return
    }
    
    // 3. 检查 App Key
    val appKey = network.appKeys.firstOrNull() ?: run {
        Log.e("MeshApp", "未找到 App Key")
        return
    }
    
    // 4. 查找设备
    val node = network.provisionedNodes.find { it.unicastAddress == address }
    if (node == null) {
        Log.e("MeshApp", "未找到设备: 0x${address.toString(16)}")
        return
    }
    
    // 5. 查找 Sensor Server 模型
    val sensorServerModelId = 0x1100
    var sensorElement: Element? = null
    
    for (element in node.elements.values) {
        if (element.meshModels.containsKey(sensorServerModelId)) {
            sensorElement = element
            break
        }
    }
    
    if (sensorElement == null) {
        Log.e("MeshApp", "设备没有 Sensor Server 模型")
        return
    }
    
    // 6. 检查模型绑定
    val model = sensorElement.meshModels[sensorServerModelId]
    if (model != null && !model.boundAppKeyIndexes.contains(appKey.keyIndex)) {
        Log.w("MeshApp", "警告: Sensor Server 未绑定 App Key ${appKey.keyIndex}")
        // 可以尝试绑定，或者提示用户
    }
    
    // 7. 发送消息
    val targetAddress = sensorElement.elementAddress
    Log.d("MeshApp", "发送 Sensor Get:")
    Log.d("MeshApp", "  目标地址: 0x${targetAddress.toString(16)}")
    Log.d("MeshApp", "  App Key Index: ${appKey.keyIndex}")
    
    try {
        MeshState.meshManagerApi.createMeshPdu(targetAddress, SensorGet(appKey, null))
        Log.d("MeshApp", "Sensor Get 已发送")
    } catch (e: Exception) {
        Log.e("MeshApp", "发送失败: ${e.message}")
    }
}
```

```kotlin
private val temperatureRequests = mutableMapOf<Int, Long>()
private val handler = Handler(Looper.getMainLooper())

fun readTemperatureWithRetry(address: Int, retryCount: Int = 0) {
    val maxRetries = 3
    val timeout = 2000L
    
    // 记录请求时间
    temperatureRequests[address] = System.currentTimeMillis()
    
    // 发送请求
    readTemperature(address)
    
    // 设置超时检查
    handler.postDelayed({
        val requestTime = temperatureRequests[address] ?: return@postDelayed
        val elapsed = System.currentTimeMillis() - requestTime
        
        if (elapsed >= timeout && retryCount < maxRetries) {
            Log.w("MeshApp", "温度读取超时，重试 ${retryCount + 1}/$maxRetries")
            readTemperatureWithRetry(address, retryCount + 1)
        }
    }, timeout)
}

// 在 parseSensorStatus 中清除请求记录
private fun parseSensorStatus(src: Int, data: ByteArray) {
```
